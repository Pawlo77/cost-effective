% ---------------------------------------------------------------------------
\begin{frame}{Problem}
    \textbf{Task:} rank 5\,000 customers by acceptance probability; contact top $k \leq 1000$.

    \vspace{0.6em}
    \textbf{Score}:
    \[
        \text{Score} = 10\,\mathrm{TP} - 5\,\mathrm{FP} - 200\,N_{\mathrm{vars}}
    \]

    \begin{itemize}
        \item 5\,000 training rows, 500 features, $\approx$50\% positive rate
        \item each declared feature costs 200 pts ($\approx$20 extra TPs to break even)
        \item break-even contact: $E[\text{contact}] = 15p - 5 > 0 \Rightarrow p > 1/3$
    \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Pipeline}
    \begin{enumerate}
        \item \textbf{Feature selection} --- 500 $\to$ ranked \texttt{top\_01}--\texttt{top\_26}
        \item \textbf{Baselines} --- no-skill floors, feature-tax demo
        \item \textbf{Modelling} --- five targeting strategies
        \item \textbf{Comparison} --- OOF business score across approaches
    \end{enumerate}

    \vspace{0.5em}
    \textbf{Strategies:}
    \begin{columns}[T,onlytextwidth]
        \column{0.48\textwidth}
        \small
        \begin{itemize}
            \item Top-$k$ profit curve
            \item EV profit targeting
            \item Rank fusion committee
        \end{itemize}
        \column{0.48\textwidth}
        \small
        \begin{itemize}
            \item Cluster split (KMeans)
            \item Segment-aware (GMM)
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Modelling approaches}
    \small
    \begin{table}
        \begin{tabular}{p{0.28\textwidth}p{0.66\textwidth}}
            \toprule
            \textbf{Approach} & \textbf{Idea} \\
            \midrule
            Top-$k$ curve &
            Compare LR / LGBM / XGB; $k^* = \arg\max_k \mathrm{Score}_k$ on OOF curve \\
            EV targeting &
            Isotonic calibration; $k = \min(k_{\mathrm{ev}}, k_{\mathrm{elbow}}, k_{\mathrm{nested}})$ \\
            Rank fusion &
            4 experts, OOF-weighted $\hat{p}$; 8-feature union penalty \\
            Cluster split &
            KMeans ($K{=}4$) + per-cluster logistic on same \texttt{top\_*} \\
            Segment-aware &
            GMM on \texttt{top\_10} (latent); logistic on \texttt{top\_03} per segment \\
            \bottomrule
        \end{tabular}
    \end{table}

\begin{frame}{Conclusions}
    \begin{itemize}
        \item \textbf{Difficult/noisy data} --- model on entire dataset did not improve F1 (PCA analysis), random .
        \item \textbf{Feature tax dominates} --- \texttt{var\_389} alone wins CV; more vars $\downarrow$ business score
        \item \textbf{Local rankers help} --- cluster split (3\,280) $\to$ segment targeting (3\,380)
    \end{itemize}

\end{frame}
