% ---------------------------------------------------------------------------
\begin{frame}{Problem}
    \textbf{Task:} rank 5\,000 customers by acceptance probability; contact top $k \leq 1000$.

    \vspace{0.6em}
    \textbf{Score}:
    \[
        \text{Score} = 10\,\mathrm{TP} - 5\,\mathrm{FP} - 200\,N_{\mathrm{vars}}
    \]

    \begin{itemize}
        \item 5\,000 training rows, 500 features, $\approx$50\% positive rate
        \item each declared feature costs 200 pts ($\approx$20 extra TPs to break even)
        \item break-even contact: $E[\text{contact}] = 15p - 5 > 0 \Rightarrow p > 1/3$
    \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Two pipelines}
    \begin{columns}[T,onlytextwidth]
        \column{0.48\textwidth}
        \textbf{Classic pipeline}
        \begin{enumerate}\small
            \item Feature selection (500 $\to$ 26)
            \item Baselines
            \item Five modelling strategies
            \item OOF comparison
        \end{enumerate}
        \vspace{0.3em}
        \small Strategies: top-$k$ curve, EV targeting, rank fusion, cluster split, segment-aware

        \column{0.48\textwidth}
        \textbf{Alternative: nested CV}
        \begin{enumerate}\small
            \item 257 prescreening strategies
            \item Inner-CV two-round search (42 models)
            \item Outer eval (rate-matched, 2 tracks)
            \item Final refit + test scoring
        \end{enumerate}
        \vspace{0.3em}
        \small Strict data-access contract; joint feature--model search
    \end{columns}
\end{frame}

\begin{frame}{Alternative approach: why nested CV?}
    \textbf{Classic pipeline result:} best OOF~3\,380 (segment targeting, 3 features).

    \vspace{0.5em}
    \textbf{Limitations:}
    \begin{itemize}
        \item prescreening fitted once on all training data $\to$ leaks fold info
        \item feature and model selection done separately, not jointly
    \end{itemize}

    \vspace{0.5em}
    \textbf{Solution --- nested $5 \times 5$ CV:}
    \begin{itemize}
        \item outer: 5 folds (4\,000 train / 1\,000 val)
        \item inner: 5 folds (3\,200 train / 800 val)
        \item all prescreening refitted from scratch inside each outer fold
        \item joint search over \emph{prescreening strategy} + \emph{model spec}
    \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Prescreening: 257 strategies}
    \textbf{8 ranking methods} across 3 families:
    \begin{columns}[T,onlytextwidth]
        \column{0.32\textwidth}
        \small\textit{Filter}
        \begin{itemize}\footnotesize
            \item Mutual information
            \item Pearson $|r|$
        \end{itemize}
        \column{0.32\textwidth}
        \small\textit{Embedded}
        \begin{itemize}\footnotesize
            \item $\ell_1$ logistic
            \item ABESS
        \end{itemize}
        \column{0.32\textwidth}
        \small\textit{Model-based}
        \begin{itemize}\footnotesize
            \item LightGBM gain
            \item EBM
            \item SPAM (sparse GAM)
            \item LambdaMART
        \end{itemize}
    \end{columns}

    \vspace{0.6em}
    \textbf{Ensemble rank score} per feature $i$ under method $j$:
    \[
        r_i = 1 - \frac{\text{rank}(s_i)-1}{p-1}, \qquad
        \bar{r}_i = \frac{1}{m}\sum_{j=1}^{m} r_i^{(j)}
    \]

    \vspace{0.3em}
    All 255 non-empty subsets $+$ 2 adaptive $=$ \textbf{257 strategies}\\
    $\times$ 16 sizes $k \in \{1,\ldots,50\}$ $=$ \textbf{4\,112 (strategies, $k$) pairs}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Inner-CV: two-round selection}
    \textbf{Round 1 --- feature screening} (5 models):
    \begin{itemize}
        \item top-300 feature sets per inner fold evaluated on 5 screening classifiers
        \item top-\textbf{60} feature strategies per outer fold advance
    \end{itemize}

    \vspace{0.4em}
    \textbf{Round 2 --- full model grid} (42 configurations):
    \begin{itemize}
        \item 60 feature strategies $\times$ 42 models across 7 families
        \item top-\textbf{20} globally selected once for all outer folds
    \end{itemize}

    \vspace{0.4em}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Outer evaluation}
    \textbf{Rate-matched scoring:}
    \begin{itemize}
        \item top-200 predictions per fold (= 20\% of 1\,000)
        \item score scaled $\times 5$ $\to$ 5\,000-obs equivalent
    \end{itemize}

    \vspace{0.4em}
    \textbf{Two parallel tracks:}
    \begin{columns}[T,onlytextwidth]
        \column{0.48\textwidth}
        \textbf{Per-fold track}
        \begin{itemize}\small
            \item 5 comparable scaled scores
            \item mean $\pm$ std over folds
        \end{itemize}
        \column{0.48\textwidth}
        \textbf{Pooled OOF track}
        \begin{itemize}\small
            \item 5\,000 disjoint predictions
            \item single deployment estimate
        \end{itemize}
    \end{columns}

    \vspace{0.5em}
    Top-10 configs by mean per-fold score $\to$ top-5 refitted on full training set and scored on test.
\end{frame}

\begin{frame}{Conclusions}
    \textbf{Classic pipeline:}
    \begin{itemize}
        \item Feature tax dominates --- V390 alone wins; more variables $\downarrow$ business score
        \item Local rankers help --- cluster split (3\,280) $\to$ segment-aware (3\,380)
        \item Noisy data: PCA analysis shows more features may introduce noise
    \end{itemize}

    \vspace{0.5em}
    \textbf{Nested CV alternative:}
    \begin{itemize}
        \item Joint search: \textbf{5\,425 OOF} \textcolor{highlight}{($+$61\% vs.\ classic)}
        \item All top-10 configs use ExtraTrees; SPAM prescreener dominates
        \item Low-variance runner-up ($\pm$64) may suggest shallower trees trade mean for stability
    \end{itemize}
\end{frame}
